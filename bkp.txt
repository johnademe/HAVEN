def get_db_connection():
    """Create and return a PostgreSQL database connection"""
    try:
        conn = psycopg2.connect(**DB_CONFIG)
        return conn
    except Exception as e:
        print(f"Database connection error: {e}")
        return None

def init_db():
    """Initialize database tables if they don't exist"""
    conn = get_db_connection()
    if not conn:
        return False
    
    cursor = conn.cursor()
    
    # Create sales table
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS sales (
            id SERIAL PRIMARY KEY,
            item VARCHAR(200) NOT NULL,
            quantity INTEGER DEFAULT 1,
            price DECIMAL(10,2) DEFAULT 0,
            total DECIMAL(10,2) DEFAULT 0,
            credit DECIMAL(10,2) DEFAULT 0,
            sale_date DATE NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    ''')
    
    # Create items table for autocomplete suggestions
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS items (
            id SERIAL PRIMARY KEY,
            item_name VARCHAR(200) UNIQUE NOT NULL,
            default_price DECIMAL(10,2),
            usage_count INTEGER DEFAULT 1,
            last_used DATE
        )
    ''')
    
    conn.commit()
    conn.close()
    return True

# ==================== Database Operations ====================
class RevenueModel:
    @staticmethod
    def add_sale(item, quantity, price, credit=0, sale_date=None):
        if sale_date is None:
            sale_date = datetime.now().strftime("%Y-%m-%d")
        
        total = quantity * price
        conn = get_db_connection()
        if not conn:
            return False
        
        cursor = conn.cursor()
        
        # Insert sale
        cursor.execute('''
            INSERT INTO sales (item, quantity, price, total, credit, sale_date)
            VALUES (%s, %s, %s, %s, %s, %s)
            RETURNING id
        ''', (item, quantity, price, total, credit, sale_date))
        
        sale_id = cursor.fetchone()[0]
        
        # Update or insert into items table for autocomplete
        cursor.execute('''
            INSERT INTO items (item_name, default_price, usage_count, last_used)
            VALUES (%s, %s, 1, %s)
            ON CONFLICT (item_name) 
            DO UPDATE SET 
                usage_count = items.usage_count + 1,
                default_price = COALESCE(EXCLUDED.default_price, items.default_price),
                last_used = EXCLUDED.last_used
        ''', (item, price, sale_date))
        
        conn.commit()
        conn.close()
        return True

    @staticmethod
    def get_sales_by_period(start_date, end_date):
        conn = get_db_connection()
        if not conn:
            return []
        
        cursor = conn.cursor()
        cursor.execute('''
            SELECT id, item, quantity, price, total, credit, sale_date
            FROM sales
            WHERE sale_date BETWEEN %s AND %s
            ORDER BY sale_date, id
        ''', (start_date, end_date))
        
        results = cursor.fetchall()
        conn.close()
        return results

    @staticmethod
    def get_all_sales():
        conn = get_db_connection()
        if not conn:
            return []
        
        cursor = conn.cursor()
        cursor.execute('''
            SELECT id, item, quantity, price, total, credit, sale_date 
            FROM sales 
            ORDER BY sale_date DESC, id DESC
        ''')
        
        results = cursor.fetchall()
        conn.close()
        return results

    @staticmethod
    def delete_sale(sale_id):
        conn = get_db_connection()
        if not conn:
            return False
        
        cursor = conn.cursor()
        cursor.execute('DELETE FROM sales WHERE id = %s', (sale_id,))
        conn.commit()
        conn.close()
        return True

    @staticmethod
    def get_item_suggestions(search_text):
        """Get item suggestions based on partial input"""
        conn = get_db_connection()
        if not conn:
            return []
        
        cursor = conn.cursor()
        cursor.execute('''
            SELECT item_name, default_price, usage_count
            FROM items
            WHERE item_name ILIKE %s
            ORDER BY usage_count DESC, last_used DESC
            LIMIT 15
        ''', (f'%{search_text}%',))
        
        results = cursor.fetchall()
        conn.close()
        return results

    @staticmethod
    def get_all_items():
        """Get all items for autocomplete"""
        conn = get_db_connection()
        if not conn:
            return []
        
        cursor = conn.cursor()
        cursor.execute('''
            SELECT item_name, default_price
            FROM items
            ORDER BY usage_count DESC, last_used DESC
        ''')
        
        results = cursor.fetchall()
        conn.close()
        return results

    @staticmethod
    def get_item_price(item_name):
        """Get default price for an item"""
        conn = get_db_connection()
        if not conn:
            return None
        
        cursor = conn.cursor()
        cursor.execute('''
            SELECT default_price FROM items WHERE item_name = %s
        ''', (item_name,))
        
        result = cursor.fetchone()
        conn.close()
        return result[0] if result else None

# ==================== Settings Dialog ====================
class SettingsDialog(QDialog):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.setWindowTitle("Database Settings")
        self.setModal(True)
        layout = QFormLayout(self)
        
        self.host_input = QLineEdit(DB_CONFIG['host'])
        self.port_input = QLineEdit(str(DB_CONFIG['port']))
        self.db_input = QLineEdit(DB_CONFIG['database'])
        self.user_input = QLineEdit(DB_CONFIG['user'])
        self.password_input = QLineEdit(DB_CONFIG['password'])
        self.password_input.setEchoMode(QLineEdit.Password)
        
        layout.addRow("Host:", self.host_input)
        layout.addRow("Port:", self.port_input)
        layout.addRow("Database:", self.db_input)
        layout.addRow("Username:", self.user_input)
        layout.addRow("Password:", self.password_input)
        
        buttons = QDialogButtonBox(QDialogButtonBox.Ok | QDialogButtonBox.Cancel)
        buttons.accepted.connect(self.save_settings)
        buttons.rejected.connect(self.reject)
        layout.addRow(buttons)
    
    def save_settings(self):
        global DB_CONFIG
        DB_CONFIG.update({
            'host': self.host_input.text(),
            'port': int(self.port_input.text()),
            'database': self.db_input.text(),
            'user': self.user_input.text(),
            'password': self.password_input.text()
        })
        self.accept()

    def on_item_text_changed(self, text):
        """Handle item text changes - show suggestions"""
        if len(text) >= 2:
            suggestions = self.model.get_item_suggestions(text)
            if suggestions:
                self.item_info_label.setText(f"💡 Found {len(suggestions)} matching items. Press Enter to auto-fill price.")
            else:
                self.item_info_label.setText("💡 New item - price will be saved for future use")
        else:
            self.item_info_label.setText("💡 Type at least 2 characters for suggestions")
    
    def auto_fill_price(self):
        """Auto-fill price when Enter is pressed"""
        item_name = self.item_input.text().strip()
        if item_name:
            price = self.model.get_item_price(item_name)
            if price is not None:
                self.price_input.setValue(price)
                self.item_info_label.setText(f"✅ Auto-filled price: ${price:.2f}")
            else:
                self.item_info_label.setText("📝 New item - enter price and it will be saved")
    
    def show_item_browser(self):
        """Show dialog to browse and select items"""
        dialog = QDialog(self)
        dialog.setWindowTitle("Browse Items")
        dialog.setMinimumSize(400, 500)
        layout = QVBoxLayout(dialog)
        
        search_box = QLineEdit()
        search_box.setPlaceholderText("Search items...")
        layout.addWidget(search_box)
        
        items_list = QTableWidget()
        items_list.setColumnCount(2)
        items_list.setHorizontalHeaderLabels(["Item Name", "Default Price"])
        
        def refresh_items():
            items = self.model.get_all_items()
            search_text = search_box.text().lower()
            if search_text:
                items = [i for i in items if search_text in i[0].lower()]
            
            items_list.setRowCount(len(items))
            for row, (name, price) in enumerate(items):
                items_list.setItem(row, 0, QTableWidgetItem(name))
                items_list.setItem(row, 1, QTableWidgetItem(f"${price:.2f}" if price else "-"))
        
        search_box.textChanged.connect(refresh_items)
        refresh_items()
        
        items_list.itemDoubleClicked.connect(lambda item: self.select_from_browser(items_list, dialog))
        
        layout.addWidget(items_list)
        
        buttons = QDialogButtonBox(QDialogButtonBox.Close)
        buttons.rejected.connect(dialog.reject)
        layout.addWidget(buttons)
        
        dialog.exec_()
    
    def select_from_browser(self, items_list, dialog):
        row = items_list.currentRow()
        if row >= 0:
            item_name = items_list.item(row, 0).text()
            price_text = items_list.item(row, 1).text().replace('$', '')
            if price_text != '-':
                self.item_input.setText(item_name)
                self.price_input.setValue(float(price_text))
                dialog.accept()
    
    def update_total_preview(self):
        total = self.qty_input.value() * self.price_input.value()
        self.total_preview_label.setText(f"💰 Total: ${total:.2f}")
    
    def add_sale(self):
        item = self.item_input.text().strip()
        if not item:
            QMessageBox.warning(self, "Input Error", "Please enter an item name.")
            return
        
        quantity = self.qty_input.value()
        price = self.price_input.value()
        credit = self.credit_input.value()
        sale_date = self.date_input.date().toString("yyyy-MM-dd")
        
        if self.model.add_sale(item, quantity, price, credit, sale_date):
            QMessageBox.information(self, "Success", 
                                   f"✅ Sale added successfully!\n\n"
                                   f"Item: {item}\n"
                                   f"Quantity: {quantity}\n"
                                   f"Price: ${price:.2f}\n"
                                   f"Total: ${quantity * price:.2f}")
            
            # Clear inputs (keep item for quick consecutive entries)
            self.qty_input.setValue(1)
            self.price_input.setValue(0)
            self.credit_input.setValue(0)
            self.date_input.setDate(QDate.currentDate())
            self.item_input.setFocus()
            self.item_input.selectAll()
            
            self.refresh_preview()
            self.refresh_all_sales()
            self.load_popular_items()
            self.load_autocomplete_items()
            self.update_stats()
        else:
            QMessageBox.critical(self, "Database Error", "Failed to add sale. Please check database connection.")
    
    def refresh_preview(self):
        sales = self.model.get_all_sales()
        self.preview_table.setRowCount(min(10, len(sales)))
        self.preview_table.setColumnCount(6)
        self.preview_table.setHorizontalHeaderLabels(["ID", "Item", "Qty", "Price", "Total", "Date"])
        
        for row, sale in enumerate(sales[:10]):
            self.preview_table.setItem(row, 0, QTableWidgetItem(str(sale[0])))
            self.preview_table.setItem(row, 1, QTableWidgetItem(sale[1]))
            self.preview_table.setItem(row, 2, QTableWidgetItem(str(sale[2])))
            self.preview_table.setItem(row, 3, QTableWidgetItem(f"${sale[3]:.2f}"))
            self.preview_table.setItem(row, 4, QTableWidgetItem(f"${sale[4]:.2f}"))
            self.preview_table.setItem(row, 5, QTableWidgetItem(str(sale[6])))
        
        self.preview_table.resizeColumnsToContents()

    def on_period_changed(self):
        is_custom = self.period_combo.currentText() == "Custom Range"
        self.custom_widgets.setVisible(is_custom)
    
    def calculate_revenue(self):
        period = self.period_combo.currentText()
        today = QDate.currentDate()
        
        if period == "Daily":
            start = today.toString("yyyy-MM-dd")
            end = start
        elif period == "Weekly":
            start_date = today.addDays(-today.dayOfWeek())
            start = start_date.toString("yyyy-MM-dd")
            end = today.toString("yyyy-MM-dd")
        elif period == "Monthly":
            start = QDate(today.year(), today.month(), 1).toString("yyyy-MM-dd")
            end = today.toString("yyyy-MM-dd")
        elif period == "Quarterly":
            quarter_month = ((today.month() - 1) // 3) * 3 + 1
            start = QDate(today.year(), quarter_month, 1).toString("yyyy-MM-dd")
            end = today.toString("yyyy-MM-dd")
        elif period == "Yearly":
            start = QDate(today.year(), 1, 1).toString("yyyy-MM-dd")
            end = today.toString("yyyy-MM-dd")
        else:  # Custom
            start = self.start_date.date().toString("yyyy-MM-dd")
            end = self.end_date.date().toString("yyyy-MM-dd")
        
        sales = self.model.get_sales_by_period(start, end)
        
        total_revenue = sum(sale[4] for sale in sales)
        total_credit = sum(sale[5] for sale in sales)
        net_revenue = total_revenue - total_credit
        
        self.total_revenue_label.setText(f"Total Revenue\n${total_revenue:,.2f}")
        self.total_credit_label.setText(f"Total Credit\n${total_credit:,.2f}")
        self.net_revenue_label.setText(f"Net Revenue\n${net_revenue:,.2f}")
        self.transaction_count_label.setText(f"Transactions\n{len(sales)}")
        
        # Populate table
        self.revenue_table.setRowCount(len(sales))
        self.revenue_table.setColumnCount(7)
        self.revenue_table.setHorizontalHeaderLabels(["ID", "Item", "Qty", "Price", "Total", "Credit", "Date"])
        
        for row, sale in enumerate(sales):
            self.revenue_table.setItem(row, 0, QTableWidgetItem(str(sale[0])))
            self.revenue_table.setItem(row, 1, QTableWidgetItem(sale[1]))
            self.revenue_table.setItem(row, 2, QTableWidgetItem(str(sale[2])))
            self.revenue_table.setItem(row, 3, QTableWidgetItem(f"${sale[3]:.2f}"))
            self.revenue_table.setItem(row, 4, QTableWidgetItem(f"${sale[4]:.2f}"))
            self.revenue_table.setItem(row, 5, QTableWidgetItem(f"${sale[5]:.2f}"))
            self.revenue_table.setItem(row, 6, QTableWidgetItem(str(sale[6])))
        
        self.revenue_table.resizeColumnsToContents()
    
    def delete_selected_sale(self):
        selected = self.revenue_table.currentRow()
        if selected >= 0:
            sale_id = int(self.revenue_table.item(selected, 0).text())
            reply = QMessageBox.question(self, "Confirm Delete", 
                                         "Are you sure you want to delete this transaction?",
                                         QMessageBox.Yes | QMessageBox.No)
            if reply == QMessageBox.Yes:
                if self.model.delete_sale(sale_id):
                    QMessageBox.information(self, "Success", "Transaction deleted successfully.")
                    self.calculate_revenue()
                    self.refresh_preview()
                    self.refresh_all_sales()
                    self.load_popular_items()
                    self.update_stats()
                else:
                    QMessageBox.critical(self, "Error", "Failed to delete transaction.")
        else:
            QMessageBox.warning(self, "Selection Error", "Please select a transaction to delete.")
